# Consumes: results (core, 144 rows), assumption_tab (from 5.2). Produces:
# format_effect_tab, two_way_anova_tables (consumed by 5.12 and the figures
# cell). Writes: thesis_tables/5_6_*.csv. No API calls.
def section_5_6(df, assumption_tab):
    heading("5.6 ANOVA AND HYPOTHESIS TESTS")
    core = df[df["cell_type"] == "core"]

    anova_rows = []
    two_way_tables = {}  # metric -> ANOVA table, reused by the Figures section
    for metric in EIGHT_DVS:
        d = dedup_config(core) if metric in DEDUP_METRICS else core
        model, tab = two_way_anova(d, metric)
        two_way_tables[metric] = tab
        report(tab, f"5_6_anova_{metric}", columns=ANOVA_PRINT_COLS, model=model,
               title=f"Two-way ANOVA (format x chunking, with interaction): "
                     f"{metric}  (n={len(d)})")

        for term in tab.index:
            if term == "Residual":
                continue
            anova_rows.append(dict(
                metric=metric, term=term, df=tab.loc[term, "df"],
                F=tab.loc[term, "F"], p=tab.loc[term, "PR(>F)"],
                partial_eta_sq=tab.loc[term, "partial_eta_sq"],
                effect_size=tab.loc[term, "effect_size"],
            ))

    anova_summary = pd.DataFrame(anova_rows)
    save_table(anova_summary, "5_6_anova_summary_all_dvs", index=False)

    subheading("Two-way ANOVA: format term across all eight DVs")
    fmt_only = anova_summary[anova_summary["term"] == "C(format)"]
    print(fmt_only[["metric", "F", "p", "partial_eta_sq"]].to_string(index=False))
    print(f"\nSignificant at alpha={ALPHA}: "
          f"{(fmt_only['p'] < ALPHA).sum()} of {len(fmt_only)} DVs.")
    print("\nPre-registered model: two-way, format x chunking, with interaction.")
    print("Assumption checks (5.2) fail on all eight DVs, so this model is a")
    print("sensitivity check. Friedman and Wilcoxon are primary. Three-way and")
    print("per-corpus models are extensions of the pre-registered specification.")

    heading("5.6 CONFLICT RESOLUTION - format term, all four combinations")
    print("{faithfulness, ragas_faithfulness} x {pre-registered two-way, the")
    print("earlier six-term model}, on the identical 144-row core frame.")
    print("two-way  = C(format) + C(chunking) + C(format):C(chunking)")
    print("six-term = two-way + C(chunk_size) + C(generator) + C(corpus)")
    say("(reproduces results/analysis/SUPERSEDED_anova_faithfulness_sixterm_legacyDV.csv)",
        indent="           ")

    model_faith2, tab_faith2 = two_way_anova(core, "faithfulness")
    model_ragas2, tab_ragas2 = two_way_anova(core, "ragas_faithfulness")
    model_faith6, tab_faith6 = six_term_anova(core, "faithfulness")
    model_ragas6, tab_ragas6 = six_term_anova(core, "ragas_faithfulness")
    save_table(tab_faith6, "5_6_anova_sixterm_faithfulness")
    save_table(tab_ragas6, "5_6_anova_sixterm_ragas_faithfulness")

    grid_rows = []
    for dv_label, tab2, tab6 in [("faithfulness", tab_faith2, tab_faith6),
                                  ("ragas_faithfulness", tab_ragas2, tab_ragas6)]:
        for model_label, tab in [("two-way", tab2), ("six-term", tab6)]:
            grid_rows.append(dict(
                dv=dv_label, model=model_label,
                F=tab.loc["C(format)", "F"], df=tab.loc["C(format)", "df"],
                resid_df=tab.loc["Residual", "df"],
                p=tab.loc["C(format)", "PR(>F)"],
                partial_eta_sq=tab.loc["C(format)", "partial_eta_sq"],
                format_sum_sq=tab.loc["C(format)", "sum_sq"],
            ))
    grid_tab = pd.DataFrame(grid_rows)
    report(grid_tab, "5_6_format_term_four_combinations", index=False,
           columns=["dv", "model", "F", "df", "resid_df", "p", "partial_eta_sq"],
           title="format term - F, df, p, partial eta squared, all four combinations")

    print("\nIdentifying F(2,130)=6.68, p=0.0017 for the format term:")
    for _, r in grid_tab.iterrows():
        match = (r["resid_df"] == 130 and abs(r["F"] - 6.68) < 0.05)
        say(f"{r['dv']:<20} x {r['model']:<9} F={r['F']:.4f}, df=(2,{r['resid_df']:.0f}), "
            f"p={fmt_p(r['p'])}  -> {'MATCH' if match else 'does not match'}", indent="  ")

    legacy_path = Path("results/analysis/SUPERSEDED_anova_faithfulness_sixterm_legacyDV.csv")
    if legacy_path.exists():
        legacy = pd.read_csv(legacy_path, index_col=0)
        if "C(format)" in legacy.index:
            leg_F, leg_p = legacy.loc["C(format)", "F"], legacy.loc["C(format)", "PR(>F)"]
            mine_F = tab_faith6.loc["C(format)", "F"]
            print(f"\n  Six-term model on `faithfulness`, recomputed here: "
                  f"F={mine_F:.4f}, p={fmt_p(tab_faith6.loc['C(format)', 'PR(>F)'])}")
            say(f"results/analysis/SUPERSEDED_anova_faithfulness_sixterm_legacyDV.csv "
                f"(the earlier run, renamed to flag it as superseded): "
                f"F={leg_F:.4f}, p={fmt_p(leg_p)}", indent="  ")
            print(f"  -> {'Recomputation matches the earlier file.' if abs(mine_F - leg_F) < 0.01 else 'DOES NOT MATCH the earlier file - investigate.'}")
    else:
        print("\n  (results/analysis/SUPERSEDED_anova_faithfulness_sixterm_legacyDV.csv "
              "not found - skipping direct file cross-check.)")

    # Decomposition: DV, model spec, or both?
    subheading("Decomposition - does the discrepancy come from the DV, "
               "the model spec, or both?")
    ss_f2 = tab_faith2.loc["C(format)", "sum_sq"]
    ss_f6 = tab_faith6.loc["C(format)", "sum_sq"]
    ss_r2 = tab_ragas2.loc["C(format)", "sum_sq"]
    ss_r6 = tab_ragas6.loc["C(format)", "sum_sq"]
    print("  format's own sum of squares, model spec held fixed, DV varied:")
    say(f"two-way :  faithfulness SS_format={ss_f2:.6f}   "
        f"ragas_faithfulness SS_format={ss_r2:.6f}", indent="    ")
    say(f"six-term:  faithfulness SS_format={ss_f6:.6f}   "
        f"ragas_faithfulness SS_format={ss_r6:.6f}", indent="    ")
    ss_faith_stable = abs(ss_f2 - ss_f6) < 1e-4
    ss_ragas_stable = abs(ss_r2 - ss_r6) < 1e-4
    print(f"\n  core is a fully balanced/orthogonal design (3 format x 3 chunking x")
    print(f"  2 chunk_size x 2 generator x 4 corpus = 144, no empty cells), so under")
    print(f"  Type II SS, format's own sum of squares should be invariant to which")
    print(f"  other orthogonal factors are added to the model. Confirmed here:")
    say(f"SS_format is {'IDENTICAL (to float precision)' if ss_faith_stable and ss_ragas_stable else 'NOT identical - design orthogonality assumption does not hold'} "
        f"between the two-way and six-term models, for both DVs.", indent="  ")
    print("\n  Consequence: the MODEL SPEC changes SIGNIFICANCE only by shrinking the")
    print("  residual - adding C(generator) [eta_sq~0.57] and C(corpus) [eta_sq~0.75]")
    say(f"removes variance from the error term (residual SS: faithfulness "
        f"{tab_faith2.loc['Residual','sum_sq']:.4f} -> {tab_faith6.loc['Residual','sum_sq']:.4f}), which inflates F for", indent="  ")
    print("  every remaining term including format, WITHOUT changing format's own")
    print("  effect. The DV changes the SIZE of the format effect itself: SS_format")
    print(f"  is {ss_f2 / ss_r2:.1f}x larger for faithfulness than for ragas_faithfulness, in")
    print("  both models - a real difference between the two constructs (5.5),")
    print("  independent of model spec.")
    print("\n  CONCLUSION: BOTH matter, for different, non-overlapping reasons.")
    print(f"    - Model spec is necessary for significance: even faithfulness's larger")
    say(f"true effect is NOT significant under the two-way spec "
        f"(F={tab_faith2.loc['C(format)','F']:.4f}, p={fmt_p(tab_faith2.loc['C(format)','PR(>F)'])}).", indent="      ")
    print(f"    - DV is necessary too: ragas_faithfulness stays non-significant even")
    say(f"under the six-term spec (F={tab_ragas6.loc['C(format)','F']:.4f}, "
        f"p={fmt_p(tab_ragas6.loc['C(format)','PR(>F)'])}) - the six-term model alone does not", indent="      ")
    print(f"      reproduce the original result on the RAGAS DV.")
    print(f"    - Only faithfulness UNDER the six-term spec reaches significance")
    print(f"      (F={tab_faith6.loc['C(format)','F']:.4f}, p={fmt_p(tab_faith6.loc['C(format)','PR(>F)'])}). Neither factor alone")
    print(f"      reproduces F(2,130)=6.68; both had to change together.")

    print("\nRESOLUTION: the pooled two-way model this thesis is pre-registered to")
    print("report (5.6 main loop above) finds no format effect on ragas_faithfulness")
    print("(F={:.4f}, p={}); that is the number to cite for H1a. The three-way and".format(
        tab_ragas2.loc["C(format)", "F"], fmt_p(tab_ragas2.loc["C(format)", "PR(>F)"])))
    print("per-corpus extensions below explain why: once corpus is modeled, format's")
    print("effect is real but reverses direction by corpus, which a model that")
    print("ignores corpus - two-way OR six-term, on either DV - cannot resolve.")

    heading("5.6 EXTENSION - THREE-WAY ANOVA (format x chunking x corpus)")
    print("Adds corpus as a third factor, plus all two-way interactions and the")
    print("three-way interaction, to the pre-registered two-way model above.")

    anova3_rows = []
    assump3_rows = []
    for metric in EIGHT_DVS:
        d = dedup_config(core) if metric in DEDUP_METRICS else core
        model3, tab3 = three_way_anova(d, metric)
        report(tab3, f"5_6_anova3way_{metric}", columns=ANOVA_PRINT_COLS, model=model3,
               title=f"Three-way ANOVA (format x chunking x corpus): "
                     f"{metric}  (n={len(d)})")

        for term in tab3.index:
            if term == "Residual":
                continue
            anova3_rows.append(dict(
                metric=metric, term=term, df=tab3.loc[term, "df"],
                F=tab3.loc[term, "F"], p=tab3.loc[term, "PR(>F)"],
                partial_eta_sq=tab3.loc[term, "partial_eta_sq"],
                effect_size=tab3.loc[term, "effect_size"],
            ))

        # Re-run assumption checks on THESE (three-way model) residuals.
        w, p_shapiro = stats.shapiro(model3.resid)
        grps = [g[metric].values for _, g in
                d.groupby(["format", "chunking", "corpus"], observed=True) if len(g) > 1]
        if len(grps) >= 2:
            lev_stat, p_levene = stats.levene(*grps)
        else:
            lev_stat, p_levene = float("nan"), float("nan")
        normal_ok = p_shapiro > ALPHA
        var_ok = (p_levene > ALPHA) if not np.isnan(p_levene) else False
        assump3_rows.append(dict(
            metric=metric, n=len(d), shapiro_W=round(w, 4), shapiro_p=p_shapiro,
            levene_stat=round(lev_stat, 4) if not np.isnan(lev_stat) else np.nan,
            levene_p=p_levene, normality_ok=normal_ok, homogeneity_ok=var_ok,
            parametric_anova_justified=normal_ok and var_ok,
        ))

    anova3_summary = pd.DataFrame(anova3_rows)
    save_table(anova3_summary, "5_6_anova3way_summary_all_dvs", index=False)

    assump3_tab = pd.DataFrame(assump3_rows)
    report(assump3_tab, "5_6_assumption_checks_threeway", index=False,
           title="Assumption checks on THREE-WAY model residuals "
                 "(format x chunking x corpus cells)")
    print("\nPer-DV verdict (three-way model):")
    for _, r in assump3_tab.iterrows():
        verdict = ("Parametric inference IS now justified for this DV." if
                   r["parametric_anova_justified"] else
                   "Not justified even with corpus - sensitivity check only, "
                   "not confirmatory.")
        print_row(r["metric"], verdict)

    heading("5.6 EXTENSION - LOGIT TRANSFORM ON BOUNDED [0,1] DVs")
    say(f"Applied to the 7 DVs above that are bounded proportions "
        f"(ctx_tokens_per_query is a token count and is excluded). logit(p) = "
        f"ln(p/(1-p)), p clipped to [{LOGIT_EPS}, {1 - LOGIT_EPS}] before")
    print("transforming - recall_at_5 and legacy faithfulness both hit exactly 1.0")
    print("in this data, where raw logit is undefined. Raw-scale ANOVA and")
    print("assumption-check results above are kept as the primary report; this")
    print("extension only checks whether the transform fixes the normality/")
    print("homogeneity violations found in 5.2 (two-way) and EXTENSION 1 (three-way).")

    raw2_lookup = assumption_tab.set_index("metric")
    raw3_lookup = assump3_tab.set_index("metric")
    logit_compare_rows = []

    for metric in BOUNDED_DVS:
        d = dedup_config(core) if metric in DEDUP_METRICS else core
        d = d.copy()
        d["_logit"] = logit(d[metric])

        model2, tab2 = two_way_anova(d, "_logit")
        report(tab2, f"5_6_anova_logit_twoway_{metric}", columns=ANOVA_PRINT_COLS,
               title=f"Logit-transformed two-way ANOVA: {metric}  (n={len(d)})")

        w2, p_sh2 = stats.shapiro(model2.resid)
        grps2 = [g["_logit"].values for _, g in
                 d.groupby(["format", "chunking"], observed=True) if len(g) > 1]
        _, p_lev2 = stats.levene(*grps2) if len(grps2) >= 2 else (float("nan"), float("nan"))
        norm2_ok, var2_ok = p_sh2 > ALPHA, (p_lev2 > ALPHA if not np.isnan(p_lev2) else False)

        model3l, tab3l = three_way_anova(d, "_logit")
        report(tab3l, f"5_6_anova_logit_threeway_{metric}", columns=ANOVA_PRINT_COLS,
               title=f"Logit-transformed three-way ANOVA: {metric}  (n={len(d)})")

        w3, p_sh3 = stats.shapiro(model3l.resid)
        grps3 = [g["_logit"].values for _, g in
                 d.groupby(["format", "chunking", "corpus"], observed=True) if len(g) > 1]
        _, p_lev3 = stats.levene(*grps3) if len(grps3) >= 2 else (float("nan"), float("nan"))
        norm3_ok, var3_ok = p_sh3 > ALPHA, (p_lev3 > ALPHA if not np.isnan(p_lev3) else False)

        raw2, raw3 = raw2_lookup.loc[metric], raw3_lookup.loc[metric]
        logit_compare_rows.append(dict(
            metric=metric, model="two-way",
            raw_shapiro_p=raw2["shapiro_p"], raw_levene_p=raw2["levene_p"],
            raw_normality_ok=bool(raw2["normality_ok"]),
            raw_homogeneity_ok=bool(raw2["homogeneity_ok"]),
            logit_shapiro_p=p_sh2, logit_levene_p=p_lev2,
            logit_normality_ok=norm2_ok, logit_homogeneity_ok=var2_ok,
            normality_restored=bool(norm2_ok and not raw2["normality_ok"]),
        ))
        logit_compare_rows.append(dict(
            metric=metric, model="three-way",
            raw_shapiro_p=raw3["shapiro_p"], raw_levene_p=raw3["levene_p"],
            raw_normality_ok=bool(raw3["normality_ok"]),
            raw_homogeneity_ok=bool(raw3["homogeneity_ok"]),
            logit_shapiro_p=p_sh3, logit_levene_p=p_lev3,
            logit_normality_ok=norm3_ok, logit_homogeneity_ok=var3_ok,
            normality_restored=bool(norm3_ok and not raw3["normality_ok"]),
        ))

    logit_compare_tab = pd.DataFrame(logit_compare_rows)
    report(logit_compare_tab, "5_6_logit_transform_assumption_comparison", index=False,
           title="Raw-scale vs logit-scale assumption checks, side by side")

    print("\nNormality-restored verdict (Shapiro-Wilk on residuals, as requested):")
    for _, r in logit_compare_tab.iterrows():
        if r["raw_normality_ok"]:
            verdict = "raw residuals were already normal - transform not needed."
        elif r["logit_normality_ok"]:
            verdict = "RESTORED - logit passes Shapiro-Wilk where raw did not."
        else:
            verdict = "NOT restored - logit still fails Shapiro-Wilk."
        label = f"{r['metric']:<26} {r['model']:<10}"
        print_row(label, verdict, label_width=len(label))

    n_restored = int(logit_compare_tab["normality_restored"].sum())
    n_total = len(logit_compare_tab)
    print()
    say(f"{n_restored}/{n_total} (metric, model) combinations had normality "
        f"restored by the logit transform.")
    print("Raw-scale results (5.2, and the two-way/three-way ANOVA tables above)")
    print("remain the primary report per the pre-registered spec; the logit-scale")
    print("tables above are kept as a sensitivity check wherever normality was")
    print("restored, and are otherwise informative but not decisive.")

    heading("5.6 EXTENSION - TWO-WAY ANOVA (format x chunking) WITHIN EACH CORPUS")
    print("Same pre-registered two-way model, refit separately per corpus so the")
    print("format and chunking effects can be inspected without being pooled")
    print("across corpora (corpus explains most of the variance - see above).")

    corpora = sorted(core["corpus"].unique())
    percorpus_rows = []
    format_effect_rows = []
    for metric in EIGHT_DVS:
        for corpus in corpora:
            core_c = core[core["corpus"] == corpus]
            d = dedup_config(core_c) if metric in DEDUP_METRICS else core_c
            model_c, tab_c = two_way_anova(d, metric)
            for term in tab_c.index:
                if term == "Residual":
                    continue
                percorpus_rows.append(dict(
                    corpus=corpus, metric=metric, term=term, n=len(d),
                    df=tab_c.loc[term, "df"], F=tab_c.loc[term, "F"],
                    p=tab_c.loc[term, "PR(>F)"],
                    partial_eta_sq=tab_c.loc[term, "partial_eta_sq"],
                    effect_size=tab_c.loc[term, "effect_size"], sig=tab_c.loc[term, "sig"],
                ))
            fterm = tab_c.loc["C(format)"]
            means = d.groupby("format", observed=True)[metric].mean()
            format_effect_rows.append(dict(
                metric=metric, corpus=corpus, F=fterm["F"], df=fterm["df"],
                p=fterm["PR(>F)"], partial_eta_sq=fterm["partial_eta_sq"], sig=fterm["sig"],
                top_format=means.idxmax(),
                **{f"mean_{f}": round(v, 4) for f, v in means.items()},
            ))

    percorpus_tab = pd.DataFrame(percorpus_rows)
    report(percorpus_tab, "5_6_anova_two_way_by_corpus", index=False,
           columns=["corpus", "metric", "term", "n", "df", "F", "p",
                    "partial_eta_sq", "sig"],
           title="Per-corpus two-way ANOVA - full term-level results "
                 "(4 corpora x 8 DVs x 3 terms)")

    # Does the format main effect hold across corpora, or does it flip?
    format_effect_tab = pd.DataFrame(format_effect_rows)
    report(format_effect_tab, "5_6_format_effect_by_corpus", index=False,
           title="Comparison - does the format main effect hold in all four "
                 "corpora, and does its top format flip?")

    print("\n(the corpus reversal grid built from this table is in section 5.12 below)")

    heading("5.6 ROBUSTNESS CHECKS - pairwise contrasts, non-parametric fallback, simple effects")

    # Holm-corrected pairwise format contrasts on ragas_faithfulness
    key = ["corpus", "chunking", "chunk_size", "generator"]
    pw = holm_pairwise(core, "ragas_faithfulness", key)
    report(pw, "5_6_pairwise_ragas_faithfulness", index=False,
           title="H1a - Holm-corrected pairwise format contrasts on ragas_faithfulness")

    flagged = pw[pw["flag"] != ""]
    if len(flagged):
        print("\nFlagged contrasts:")
        for _, r in flagged.iterrows():
            print(f"  {r['a']} vs {r['b']}: mean diff={r['mean_diff']:+.4f}, "
                  f"p_holm={fmt_p(r['p_holm'])}  -> {r['flag']}")
    else:
        print()
        say("No contrast is both statistically significant and practically "
            "negligible (|mean diff| < 0.01).")

    # Friedman/Wilcoxon fallback where assumptions fail
    subheading("Non-parametric fallback (Friedman across formats) - "
               "run for every DV whose assumptions failed in 5.2")
    fallback_rows = []
    for _, r in assumption_tab.iterrows():
        metric = r["metric"]
        if r["parametric_anova_justified"]:
            continue
        d = dedup_config(core) if metric in DEDUP_METRICS else core
        key = ["corpus", "chunking", "chunk_size"] if metric in DEDUP_METRICS \
            else ["corpus", "chunking", "chunk_size", "generator"]
        wide = d.pivot_table(index=key, columns="format", values=metric,
                              observed=True).dropna()
        if len(wide) >= 3 and wide.shape[1] >= 3:
            st, p = stats.friedmanchisquare(*[wide[c].values for c in wide.columns])
            ranks = wide.rank(axis=1, ascending=False).mean().sort_values()
            print(f"\n  {metric}: chi-sq={st:.4f}  df={wide.shape[1]-1}  p={fmt_p(p)}  "
                  f"({len(wide)} matched blocks)")
            print("    mean ranks (1=best): " +
                  ", ".join(f"{f}={v:.3f}" for f, v in ranks.items()))
            fallback_rows.append(dict(metric=metric, chi2=st, df=wide.shape[1] - 1,
                                       p=p, n_blocks=len(wide),
                                       best_format=ranks.index[0]))
        else:
            print(f"\n  {metric}: insufficient matched blocks/formats for Friedman "
                  f"(have {len(wide)} blocks).")
    fallback_tab = pd.DataFrame(fallback_rows)
    if len(fallback_tab):
        save_table(fallback_tab, "5_6_friedman_fallback", index=False)
    else:
        print("\n  No DV required the fallback (all DVs met parametric assumptions "
              "in 5.2).")

    # Simple-effects analysis for crossover detection
    subheading("Simple-effects analysis (crossover detection): format ranking "
               "within each chunking level, per DV")
    crossover_rows = []
    for metric in EIGHT_DVS:
        d = dedup_config(core) if metric in DEDUP_METRICS else core
        best = {}
        detail = []
        for ch, g in d.groupby("chunking", observed=True):
            grps = [s[metric].values for _, s in g.groupby("format", observed=True) if len(s)]
            if len(grps) < 2 or all(len(x) < 2 for x in grps):
                continue
            F, p = stats.f_oneway(*grps)
            means = g.groupby("format", observed=True)[metric].mean().sort_values(ascending=False)
            best[ch] = means.index[0]
            detail.append((ch, F, p, " > ".join(means.index)))
        crossover = len(set(best.values())) > 1
        print(f"\n  {metric}:")
        for ch, F, p, ranking in detail:
            print(f"    {ch:<10} F={F:.3f}  p={fmt_p(p)}  ranking: {ranking}")
        print(f"    -> {'CROSSOVER (best format differs by chunking)' if crossover else 'no crossover (same format wins in every chunking level)'}")
        crossover_rows.append(dict(metric=metric, crossover_detected=crossover,
                                    best_format_per_chunking=str(best)))
    crossover_tab = pd.DataFrame(crossover_rows)
    save_table(crossover_tab, "5_6_simple_effects_crossover", index=False)

    return format_effect_tab, two_way_tables


format_effect_tab, two_way_anova_tables = section_5_6(results, assumption_tab)
