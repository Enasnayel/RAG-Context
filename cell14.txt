def save_table(df, name, index=True):
    path = TABLES_DIR / f"{name}.csv"
    df.to_csv(path, index=index)
    _saved.append(name)


def fmt_p(p):
    if p is None or (isinstance(p, float) and np.isnan(p)):
        return "NA"
    if p < 0.0001:
        return f"{p:.2e}"
    return f"{p:.4f}"


ANOVA_PRINT_COLS = ["df", "F", "PR(>F)", "partial_eta_sq", "sig"]


def report(table, name, title=None, index=True, columns=None, model=None):
    """Print a table narrowly enough for the PDF export, then save the full one.

    The printed view may drop columns to fit the page; the saved CSV never does.
    """
    if title:
        subheading(title)

    printable = table.copy()
    for column in printable.columns:
        if not isinstance(column, str):
            continue
        # p-value columns follow one of two naming styles in this notebook:
        # "p" / "PR(>F)" / "p_raw" / "p_holm", or "<stage>_p" (shapiro_p,
        # raw_levene_p, logit_shapiro_p, ...).
        is_p_value = (column in ("p", "PR(>F)") or column.startswith("p_")
                     or column.endswith("_p"))
        if is_p_value:
            printable[column] = printable[column].apply(fmt_p)

    if columns:
        keep = [c for c in columns if c in printable.columns]
        printable = printable[keep]

    print(printable.round(4).to_string(index=index, line_width=CONSOLE_W))

    if model is not None:
        print(f"R-sq = {model.rsquared:.4f}   adj. R-sq = {model.rsquared_adj:.4f}")

    save_table(table, name, index=index)      # saves the FULL table, silently


def print_row(label, text, label_width=26, total_width=CONSOLE_W):
    """Print a label and its text, wrapping the text under a hanging indent.

    The label prefix has to be passed to textwrap as initial_indent, not
    prepended afterwards - otherwise textwrap wraps the text alone to
    total_width and the prefix pushes the finished first line back over it.
    """
    prefix = f"  {label:<{label_width}} "
    indent = " " * len(prefix)
    wrapped = textwrap.fill(text, width=total_width,
                            initial_indent=prefix, subsequent_indent=indent)
    print(wrapped)


def dedup_config(df, keep="first"):
    """One row per (corpus, format, chunking, chunk_size) config.

    Use for any column in DEDUP_METRICS - those are written identically to
    both generator rows, so failing to dedup double counts every observation.
    """
    return df.drop_duplicates(subset=CONFIG_KEY, keep=keep).copy()


def eta_squared(anova_table):
    """Partial eta-squared: SS_effect / (SS_effect + SS_residual)."""
    ss_r = anova_table.loc["Residual", "sum_sq"]
    return {t: anova_table.loc[t, "sum_sq"] / (anova_table.loc[t, "sum_sq"] + ss_r)
            for t in anova_table.index if t != "Residual"}


ETA_SQ_LARGE, ETA_SQ_MEDIUM, ETA_SQ_SMALL = 0.14, 0.06, 0.01


def effect_size_label(eta_sq):
    """Cohen's conventional bands for partial eta-squared."""
    if pd.isna(eta_sq):
        return ""
    if eta_sq >= ETA_SQ_LARGE:
        return "large"
    if eta_sq >= ETA_SQ_MEDIUM:
        return "medium"
    if eta_sq >= ETA_SQ_SMALL:
        return "small"
    return "negligible"


def sig_stars(p):
    """Conventional significance markers for a p-value."""
    if pd.isna(p):
        return ""
    if p < 0.001:
        return "***"
    if p < 0.01:
        return "**"
    if p < ALPHA:
        return "*"
    return "ns"


def logit(series, eps=LOGIT_EPS):
    """Logit transform for a [0,1]-bounded series, clipped to (eps, 1-eps).

    recall_at_5 and legacy faithfulness both hit exactly 1.0 in this data,
    where raw logit is undefined; eps=1e-4 is a fixed small-sample correction,
    applied uniformly across all bounded DVs for comparability.
    """
    clipped = series.clip(lower=eps, upper=1 - eps)
    return np.log(clipped / (1 - clipped))


def two_way_anova(df, metric):
    """format x chunking two-way ANOVA with interaction, Type II SS.

    Returns (model, table). table has columns F, df, PR(>F), partial_eta_sq.
    """
    d = df.copy()
    d["format"] = d["format"].astype("category")
    d["chunking"] = d["chunking"].astype("category")
    model = ols(f"Q('{metric}') ~ C(format) + C(chunking) + C(format):C(chunking)",
                data=d).fit()
    tab = sm.stats.anova_lm(model, typ=2)
    tab["partial_eta_sq"] = pd.Series(eta_squared(tab))
    tab["effect_size"] = tab["partial_eta_sq"].apply(effect_size_label)
    tab["sig"] = tab["PR(>F)"].apply(sig_stars)
    return model, tab


def three_way_anova(df, metric):
    """format x chunking x corpus three-way ANOVA, all interactions, Type II SS.

    Extension of the pre-registered two-way (format x chunking) model: adds
    corpus as a third factor plus its interactions, since corpus is by far
    the largest source of variance in these DVs (see 5.6) and is omitted from
    the two-way model. Type II SS kept for consistency with two_way_anova.
    Returns (model, table).
    """
    d = df.copy()
    for c in ("format", "chunking", "corpus"):
        d[c] = d[c].astype("category")
    model = ols(f"Q('{metric}') ~ C(format) * C(chunking) * C(corpus)", data=d).fit()
    tab = sm.stats.anova_lm(model, typ=2)
    tab["partial_eta_sq"] = pd.Series(eta_squared(tab))
    tab["effect_size"] = tab["partial_eta_sq"].apply(effect_size_label)
    tab["sig"] = tab["PR(>F)"].apply(sig_stars)
    return model, tab


def six_term_anova(df, metric):
    """format + chunking + format:chunking + chunk_size + generator + corpus,
    Type II SS - no other interactions.

    Reproduces the model the earlier analysis fit (see
    results/analysis/SUPERSEDED_anova_faithfulness_sixterm_legacyDV.csv): the
    pre-registered two-way model plus three more main effects, but none of
    THEIR interactions. Used only for direct comparison against that earlier
    run in the conflict-resolution block below, not as a competing formal
    test. Returns (model, table).
    """
    d = df.copy()
    for c in ("format", "chunking", "generator", "corpus"):
        d[c] = d[c].astype("category")
    d["chunk_size"] = d["chunk_size"].astype(str).astype("category")
    formula = (f"Q('{metric}') ~ C(format) + C(chunking) + C(format):C(chunking) "
               f"+ C(chunk_size) + C(generator) + C(corpus)")
    model = ols(formula, data=d).fit()
    tab = sm.stats.anova_lm(model, typ=2)
    tab["partial_eta_sq"] = pd.Series(eta_squared(tab))
    tab["effect_size"] = tab["partial_eta_sq"].apply(effect_size_label)
    tab["sig"] = tab["PR(>F)"].apply(sig_stars)
    return model, tab


def holm_pairwise(df, metric, key_cols):
    """Holm-corrected pairwise format contrasts, paired within key_cols.

    Wilcoxon signed-rank on matched blocks (falls back to paired t-test if
    Wilcoxon can't be computed, e.g. all differences zero), Mann-Whitney U if
    blocks can't be matched. Effect size is Cohen's d_z (paired mean
    difference / SD of the differences).
    """
    fmts = sorted(df["format"].unique())
    tests = []
    for i, a in enumerate(fmts):
        for b in fmts[i + 1:]:
            da = df[df["format"] == a].set_index(key_cols)[metric]
            db = df[df["format"] == b].set_index(key_cols)[metric]
            common = da.index.intersection(db.index)
            if len(common) >= 3:
                x, y = da.loc[common].values, db.loc[common].values
                diffs = x - y
                try:
                    st, p = stats.wilcoxon(x, y)
                    test = "Wilcoxon"
                except ValueError:
                    st, p = stats.ttest_rel(x, y)[:2]
                    test = "paired t"
                d = float(np.mean(diffs))
                sd = float(np.std(diffs, ddof=1))
                dz = d / sd if sd > 1e-12 else float("nan")
                n = len(common)
            elif len(da) >= 2 and len(db) >= 2:
                st, p = stats.mannwhitneyu(da.values, db.values, alternative="two-sided")
                test, d = "Mann-Whitney", float(da.mean() - db.mean())
                pooled_sd = np.sqrt((da.var(ddof=1) + db.var(ddof=1)) / 2)
                dz = d / pooled_sd if pooled_sd > 1e-12 else float("nan")
                n = len(da) + len(db)
            else:
                continue
            tests.append(dict(a=a, b=b, n=n, mean_diff=d, effect_size_dz=dz,
                               p_raw=p, test=test))
    if not tests:
        return pd.DataFrame()
    tests.sort(key=lambda t: t["p_raw"])
    m, prev = len(tests), 0.0
    for i, t in enumerate(tests):
        prev = t["p_holm"] = max(prev, min(1.0, (m - i) * t["p_raw"]))
        t["sig"] = sig_stars(prev)
        t["negligible_magnitude"] = abs(t["mean_diff"]) < 0.01
        t["flag"] = ("SIGNIFICANT BUT PRACTICALLY NEGLIGIBLE"
                     if t["sig"] != "ns" and t["negligible_magnitude"] else "")
    return pd.DataFrame(tests)

# Everything printed from here through the end of section 28 (Limitations
# data) is written to thesis_numbers.txt as well as displayed below each
# cell, matching the log thesis_analysis.py produced. Reassigning sys.stdout
# itself (as thesis_analysis.py does via redirect_stdout, since it runs as a
# plain script) breaks ipykernel's per-cell output capture - Jupyter routes
# each cell's stdout to that cell's own output area via the real OutStream
# object, and swapping it out silently drops every subsequent cell's display
# output even though the underlying prints still happen. Shadowing the
# builtin print() instead leaves sys.stdout untouched, so normal notebook
# display keeps working, and every call also appends to thesis_numbers.txt.
#
# Scope: print is shadowed as a notebook-global name (builtins.print itself
# is untouched) from here through the end of the Limitations cell (section
# 28), where it is explicitly restored. Every print() call in sections
# 15-28 below goes through this shadowed version, not the real builtin.
import builtins

_tee_file = OUT_TXT.open("w", encoding="utf-8")
_builtin_print = builtins.print


def print(*args, **kwargs):
    _builtin_print(*args, **kwargs)
    if kwargs.get("file") in (None, sys.stdout):
        sep = kwargs.get("sep", " ")
        end = kwargs.get("end", "\n")
        _tee_file.write(sep.join(str(a) for a in args) + end)
        _tee_file.flush()
